        /// <summary>
        /// Batched forward pass for efficient training.
        /// Processes multiple samples in parallel on GPU.
        /// </summary>
        public (float[][,] melOutputs, float[][] stopTokens, float loss) ForwardBatch(
            int[][] textTokensBatch,
            float[][,] targetMelBatch,
            int[] speakerIds)
        {
            int batchSize = textTokensBatch.Length;
            var melOutputs = new float[batchSize][,];
            var stopTokensList = new float[batchSize][];
            float totalLoss = 0f;

            // Process batch in parallel using GPU streams (simplified: sequential for now)
            // Full implementation would use CUDA streams for true parallel execution
            for (int b = 0; b < batchSize; b++)
            {
                var (melOutput, stopTokens, _) = Forward(textTokensBatch[b], targetMelBatch[b], speakerIds[b]);
                melOutputs[b] = melOutput;
                stopTokensList[b] = stopTokens;

                if (targetMelBatch[b] != null)
                {
                    totalLoss += ComputeLoss(melOutput, targetMelBatch[b], stopTokens);
                }
            }

            return (melOutputs, stopTokensList, totalLoss / batchSize);
        }
